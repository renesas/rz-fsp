OpenH264
RZA » Modules » Graphics

OpenH264 port for RZ MPUs.

Overview

The OpenH264 module enables H.264 functionality in FreeRTOS environment. It is implemented using FreeRTOS+POSIX to help translate POSIX calls to native FreeRTOS operations.

Note
FSP does not provide any interfaces to user. Consult the detail of OpenH264 at https://www.openh264.org/index.html for further information.

Features

The OpenH264 module provides decoder features:

  • Annex B byte stream input
  • YUV 4:2:0 planar output

Encoder features are currently not supported

Configuration

The OpenH264 module does not provide any configuration

Usage Notes

Project Creation

C++ project creation is required when using OpenH264

Limitations

Before building the project, user need to configure include paths to the following folder by manual: rz/freertos/Lab-Project-FreeRTOS-POSIX and rz/openh264/openh264

Steps to configure include paths:

  1. Right click the target folder (Lab-Project-FreeRTOS-POSIX or openh264)
  2. Select Properites from the context menu
  3. Navigate to C/C++ Build -> Settings
  4. Locate the following compiler sections:
    • Cross ARM GNU Assembler -> Includes
    • Cross ARM C Compiler -> Includes
    • Cross ARM C++ Compiler -> Includes
  5. In each compiler section's "Include paths" field, add the following paths at the bottom of include paths list:
    "${workspace_loc:/${ProjName}/rz/freertos/Lab-Project-FreeRTOS-POSIX/FreeRTOS-Plus-POSIX/include}"
    "${workspace_loc:/${ProjName}/rz/freertos/Lab-Project-FreeRTOS-POSIX/FreeRTOS-Plus-POSIX/include/portable}"
    "${workspace_loc:/${ProjName}/rz/freertos/Lab-Project-FreeRTOS-POSIX/include}"
    "${workspace_loc:/${ProjName}/rz/freertos/Lab-Project-FreeRTOS-POSIX/include/private}"
    "${workspace_loc:/${ProjName}/rz/freertos/Lab-Project-FreeRTOS-POSIX/include/FreeRTOS_POSIX}"

Examples

Basic Example

This is a basic example demonstrating the initialization and decoding an H.264 file by using OpenH264

#define MAX_NALS (1024 * 10)
typedef struct
{
uint8_t * data;
size_t size;
} NALUnit;
NALUnit nal_list[MAX_NALS];
static uint32_t nal_count;
/* Parse H.264 Annex-B format stream into individual NAL units */
int h264_parse_nal_units (uint8_t * buffer, uint32_t buffer_size) {
uint32_t pos = 0;
uint32_t nal_start = 0;
bool found_first_nal = false;
nal_count = 0;
while (pos <= buffer_size - 4)
{
bool is_start_code = false;
uint32_t start_code_size = 0;
/* Check for 4-byte start code (0x00000001) */
if (memcmp(&buffer[pos], "\x00\x00\x00\x01", 4) == 0)
{
is_start_code = true;
start_code_size = 4;
}
/* Check for 3-byte start code (0x000001) */
else if (memcmp(&buffer[pos], "\x00\x00\x01", 3) == 0)
{
is_start_code = true;
start_code_size = 3;
}
if (is_start_code)
{
if (found_first_nal)
{
if (nal_count >= MAX_NALS)
{
return 1;
}
nal_list[nal_count].data = &buffer[nal_start];
nal_list[nal_count].size = pos - nal_start;
nal_count++;
}
nal_start = pos; /* Include start code in NAL data */
pos += start_code_size;
found_first_nal = true;
}
else
{
pos++;
}
}
/* Handle the last NAL unit */
if (found_first_nal && (nal_start < buffer_size))
{
if (nal_count >= MAX_NALS)
{
return 1;
}
nal_list[nal_count].data = &buffer[nal_start];
nal_list[nal_count].size = buffer_size - nal_start;
nal_count++;
}
return 0;
}
static ISVCDecoder * pDecoder = NULL;
/* Initialize OpenH264 decoder */
int h264_start_decode_task (void)
{
if ((WelsCreateDecoder(&pDecoder) != 0) || (pDecoder == NULL))
{
printf("OpenH264 decoder creation failed\n");
return 1;
}
SDecodingParam sDecParam;
memset(&sDecParam, 0, sizeof(SDecodingParam));
sDecParam.uiTargetDqLayer = (uint8_t) -1;
sDecParam.eEcActiveIdc = ERROR_CON_FRAME_COPY_CROSS_IDR;
sDecParam.sVideoProperty.eVideoBsType = VIDEO_BITSTREAM_DEFAULT;
sDecParam.sVideoProperty.size = sizeof(sDecParam.sVideoProperty);
if (pDecoder->Initialize(&sDecParam) != 0)
{
printf("OpenH264 initialization failed\n");
WelsDestroyDecoder(pDecoder);
return 1;
}
return 0;
}
/* Stop OpenH264 decoder */
void h264_stop_decode_task (void)
{
pDecoder->Uninitialize();
WelsDestroyDecoder(pDecoder);
}
/* Decode H.264 frame data and display the result */
int h264_decode_task (uint8_t * h264_data, uint32_t h264_data_size)
{
/* Parse NAL units */
if (h264_parse_nal_units(h264_data, h264_data_size))
{
printf("Decoder error: Too large number of NAL\n");
return 1;
}
SBufferInfo sDstBufInfo;
memset(&sDstBufInfo, 0, sizeof(sDstBufInfo));
uint8_t * yuv[3] = {NULL, NULL, NULL};
DECODING_STATE state = dsErrorFree;
/* Start decoding each NAL unit */
for (uint32_t i = 0; i < nal_count; i++)
{
state = pDecoder->DecodeFrameNoDelay(nal_list[i].data, (int) nal_list[i].size, yuv, &sDstBufInfo);
switch (state)
{
case dsErrorFree:
{
display_yuv_frame(yuv[0],
yuv[1],
yuv[2],
(uint16_t) sDstBufInfo.UsrData.sSystemBuffer.iWidth,
(uint16_t) sDstBufInfo.UsrData.sSystemBuffer.iHeight,
sDstBufInfo.UsrData.sSystemBuffer.iStride[0],
sDstBufInfo.UsrData.sSystemBuffer.iStride[1]);
break;
}
case dsNoParamSets:
{
printf("Decoder error: missing SPS/PPS\n");
break;
}
case dsBitstreamError:
{
printf("Decoder error: bitstream corrupted NAL\n");
break;
}
case dsRefLost:
{
printf("Decoder warning: lost reference frame\n");
break;
}
case dsInvalidArgument:
{
printf("Decoder error: invalid argument\n");
break;
}
default:
{
break;
}
}
}
return state;
}
/* Copy decoded YUV frame data to display framebuffers */
void display_yuv_frame (const uint8_t * Y,
const uint8_t * U,
const uint8_t * V,
uint16_t video_width,
uint16_t video_height,
uint32_t Y_Stride,
uint32_t UV_Stride)
{
uint8_t * p_buff_0;
uint8_t * p_buff_1;
uint8_t * p_buff_2;
const uint8_t * p_data_0;
const uint8_t * p_data_1;
const uint8_t * p_data_2;
uint16_t height;
p_buff_0 = (uint8_t *) fb_background[0];
p_buff_1 = (uint8_t *) fb_background_CB[0];
p_buff_2 = (uint8_t *) fb_background_CR[0];
p_data_0 = Y;
p_data_1 = U;
p_data_2 = V;
/* Copy Y plane */
for (height = 0; height < video_height; height++)
{
memcpy(p_buff_0, p_data_0, video_width);
p_buff_0 += video_width;
p_data_0 += Y_Stride;
}
/* Copy U and V planes */
for (height = 0; height < video_height / 2; height++)
{
memcpy(p_buff_1, p_data_1, video_width / 2);
memcpy(p_buff_2, p_data_2, video_width / 2);
p_buff_1 += video_width / 2;
p_buff_2 += video_width / 2;
p_data_1 += UV_Stride;
p_data_2 += UV_Stride;
}
}
/* This is the User Thread for the EP */
void OpenH264_thread_entry (void * pvParameters)
{
FSP_PARAMETER_NOT_USED(pvParameters);
/* TODO: Perform any initialization necessary for the physical display. */
if (0 != h264_start_decode_task())
{
printf("Open H.264 start failed.\r\n");
}
while (1)
{
/* Process video data */
h264_decode_task(&video_h264_data[0], video_h264_finish - video_h264_data);
vTaskDelay(100); /* interval between next video play */
}
